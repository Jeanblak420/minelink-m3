SQL to Create run_sql Function:

-- Create a function to execute SQL queries
-- This function is designed to be used with the Supabase RPC API
-- It requires superuser privileges to execute arbitrary SQL
-- SECURITY DEFINER means it will run with the privileges of the user who created it

-- Drop the function if it already exists
DROP FUNCTION IF EXISTS run_sql(text);

-- Create the function
CREATE OR REPLACE FUNCTION run_sql(query text)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  result JSONB;
BEGIN
  -- Execute the query and capture the result
  EXECUTE 'SELECT jsonb_agg(row_to_json(t)) FROM (' || query || ') t' INTO result;
  
  -- If the query doesn't return any rows, return an empty array
  IF result IS NULL THEN
    RETURN '[]'::JSONB;
  END IF;
  
  RETURN result;
EXCEPTION
  WHEN OTHERS THEN
    -- Return error information as JSON
    RETURN jsonb_build_object(
      'error', true,
      'message', SQLERRM,
      'detail', SQLSTATE,
      'query', query
    );
END;
$$;

-- Grant execute permission to authenticated users
GRANT EXECUTE ON FUNCTION run_sql(text) TO authenticated;

-- Add a comment to the function
COMMENT ON FUNCTION run_sql(text) IS 'Executes a SQL query and returns the result as JSON. Requires superuser privileges.';
